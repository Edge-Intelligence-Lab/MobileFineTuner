cmake_minimum_required(VERSION 3.16)
project(mobilefinetuner_checkpoint_resume LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ANDROID)
    set(_test_default OFF)
else()
    set(_test_default ON)
endif()
option(BUILD_TESTS "Build checkpoint/resume native tests" ${_test_default})
if(BUILD_TESTS)
    enable_testing()
endif()
add_subdirectory(operators)
add_executable(mft_qwen_resume qwen_lora_finetune/src/main.cpp)
target_link_libraries(mft_qwen_resume PRIVATE operators)
add_executable(mft_gemma_resume gemma_3_270m_lora_finetune/src/benchmark_memory.cpp)
target_link_libraries(mft_gemma_resume PRIVATE operators)
if(ANDROID)
    set(MFT_ANDROID_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android/app/src/main/jniLibs/${ANDROID_ABI}"
        CACHE PATH "Android directory for the trainer executables and JNI process owner")
    add_library(mft_owned_process SHARED android/app/src/main/cpp/native_owned_process.cpp)
    set_target_properties(mft_owned_process PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${MFT_ANDROID_LIB_DIR}")
    foreach(name qwen gemma)
        add_dependencies(mft_${name}_resume mft_owned_process)
        set_target_properties(mft_${name}_resume PROPERTIES
            OUTPUT_NAME libmft_${name} SUFFIX .so
            RUNTIME_OUTPUT_DIRECTORY "${MFT_ANDROID_LIB_DIR}")
    endforeach()
endif()
